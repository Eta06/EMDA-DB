use database_engine::{BatchOperation, Store};
use std::alloc::{GlobalAlloc, Layout, System};
use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering::Relaxed};

struct CountingAllocator;
static COUNTING: AtomicBool = AtomicBool::new(false);
static CALLS: AtomicUsize = AtomicUsize::new(0);
static BYTES: AtomicUsize = AtomicUsize::new(0);

fn count(size: usize) {
    if COUNTING.load(Relaxed) {
        CALLS.fetch_add(1, Relaxed);
        BYTES.fetch_add(size, Relaxed);
    }
}

unsafe impl GlobalAlloc for CountingAllocator {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        count(layout.size());
        unsafe { System.alloc(layout) }
    }

    unsafe fn alloc_zeroed(&self, layout: Layout) -> *mut u8 {
        count(layout.size());
        unsafe { System.alloc_zeroed(layout) }
    }

    unsafe fn realloc(&self, pointer: *mut u8, layout: Layout, size: usize) -> *mut u8 {
        count(size);
        unsafe { System.realloc(pointer, layout, size) }
    }

    unsafe fn dealloc(&self, pointer: *mut u8, layout: Layout) {
        unsafe { System.dealloc(pointer, layout) }
    }
}

#[global_allocator]
static ALLOCATOR: CountingAllocator = CountingAllocator;

fn main() -> std::io::Result<()> {
    let mode = std::env::args().nth(1).unwrap();
    let root = tempfile::tempdir()?;
    let store = Store::open(root.path())?;
    let path = root.path().join("records.log");
    let length = std::fs::metadata(&path)?.len();
    let store = if mode == "log-rejected" {
        drop(store);
        Store::open_with_log_limit(root.path(), length)?
    } else {
        store
    };
    if mode == "group-rejected" {
        store.set_group_commit_limits(0, 0)?;
    }
    let value = vec![7; 4 * 1024 * 1024];
    let operations = [BatchOperation::Put {
        key: b"data",
        value: &value,
    }];
    COUNTING.store(true, Relaxed);
    let start = std::time::Instant::now();
    let result = if mode == "transaction" {
        store.transaction(|tx| tx.put(b"data", &value))
    } else {
        store.write_batch(&operations)
    };
    COUNTING.store(false, Relaxed);
    let elapsed_us = start.elapsed().as_micros();
    let rejected = mode.ends_with("-rejected");
    assert_eq!(result.is_err(), rejected);
    if rejected {
        assert_eq!(std::fs::metadata(&path)?.len(), length);
        assert_eq!(store.get(b"data")?, None);
    } else {
        drop(store);
        let store = Store::open(root.path())?;
        assert_eq!(store.get(b"data")?, Some(value));
    }
    println!(
        "mode={mode} calls={} requested_bytes={} elapsed_us={elapsed_us}",
        CALLS.load(Relaxed),
        BYTES.load(Relaxed)
    );
    Ok(())
}
